Розрахункова робота — варіант №15
Завдання №1. Обчислення суми математичного ряду
Умова завдання
Обчислити суму ряду:
\[
S=\sum_{n=1}^{N}(-1)^{n+1}\frac{a^n}{n},
\]
де a та N вводяться з клавіатури, N — ціле число, \(N \geq 1\).
Оголошення змінних
double a, S = 0.0, power = 1.0;
int N, n;
Введення та перевірка даних
printf("Enter a: ");
scanf("%lf", &a);

printf("Enter N (N >= 1): ");
scanf("%d", &N);

if (N < 1) {
    printf("Error: N must be >= 1.\n");
    return 1;
}
Обчислення суми ряду
Знак доданка визначається за допомогою n % 2:
- якщо n непарне, доданок додається до суми;
- якщо n парне, доданок віднімається від суми.
for (n = 1; n <= N; n++) {
    power *= a;

    if (n % 2 == 1) {
        S += power / n;
    } else {
        S -= power / n;
    }
}

printf("The result is: %.6lf\n", S);
Результат роботи
Після завершення циклу програма виводить суму S з точністю до шести знаків після коми.
Завдання №2. Сортування рядків і стовпців квадратної матриці
Оголошення змінних
Для роботи з матрицями використовуються такі змінні:
- A[20][20] — початкова квадратна матриця;
- B[20][20] — копія матриці, у якій виконується сортування;
- N — розмір матриць;
- temp — тимчасова змінна для обміну елементів.
int A[20][20], B[20][20];
int N, temp;
Максимальний розмір матриці — \(20 \times 20\).
Ініціалізація генератора та введення розміру
srand(time(NULL));

printf("Введіть розмір матриці N (1-20): ");
scanf("%d", &N);

if (N < 1 || N > 20) {
    printf("Помилка! N повинно бути від 1 до 20.\n");
    return 1;
}
Заповнення матриці та копіювання
for (int i = 0; i < N; i++) {
    for (int j = 0; j < N; j++) {
        A[i][j] = 30 + rand() % 26;
        B[i][j] = A[i][j];
    }
}
Виведення початкової матриці
printf("\nПочаткова матриця A:\n");
for (int i = 0; i < N; i++) {
    for (int j = 0; j < N; j++) {
        printf("%5d", A[i][j]);
    }
    printf("\n");
}
Сортування рядків за зростанням
for (int i = 0; i < N; i++) {
    for (int j = 0; j < N - 1; j++) {
        for (int k = 0; k < N - j - 1; k++) {
            if (B[i][k] > B[i][k + 1]) {
                temp = B[i][k];
                B[i][k] = B[i][k + 1];
                B[i][k + 1] = temp;
            }
        }
    }
}
Сортування стовпців за зростанням
for (int j = 0; j < N; j++) {
    for (int i = 0; i < N - 1; i++) {
        for (int k = 0; k < N - i - 1; k++) {
            if (B[k][j] > B[k + 1][j]) {
                temp = B[k][j];
                B[k][j] = B[k + 1][j];
                B[k + 1][j] = temp;
            }
        }
    }
}
Виведення результату
printf("\nВідсортована матриця B:\n");
for (int i = 0; i < N; i++) {
    for (int j = 0; j < N; j++) {
        printf("%5d", B[i][j]);
    }
    printf("\n");
}
